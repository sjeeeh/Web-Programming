let gender=document.getElementByName("gender");
let selected="";
for(let i=0;i<gender.length;i++){
  if(gender[i].checked){
    selected=gender[i].value;
    return true;
  }
}
if(selected==""){
  alert("Select Gender");
  return false;
}
